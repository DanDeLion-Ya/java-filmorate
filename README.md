# java-filmorate
Template repository for Filmorate project.

![Была выполнена схема Базы Данных приложения filmorate](/images/filmorate_sql_origin.png)

## Ниже располагаю код запросов для создания таблиц и связей между ними.

CREATE TABLE films(
id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
name VARCHAR(50) NOT NULL,
description VARCHAR(200),
releaseDate DATE,
duration INTEGER,
rating_id INTEGER REFERENCES mpa(id))

CREATE TABLE genre(
id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
name VARCHAR(20) NOT NULL
)

CREATE TABLE film_genre(
genre_id INTEGER REFERENCES genre(id),
film_id INTEGER REFERENCES films(id),
PRIMARY KEY(genre_id, film_id)
)

CREATE TABLE mpa(
id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
name VARCHAR(20) NOT NULL)

CREATE TABLE users(
id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
email VARCHAR(100) NOT NULL,
login VARCHAR(100) NOT NULL,
name VARCHAR(100),
birthday DATE)

CREATE TABLE friendship(
status VARCHAR(20) CHECK(status IN ('Подтвержденно', 'Неподтвержденно')),
user_id INTEGER REFERENCES users(id),
friend_id INTEGER REFERENCES users(id),
PRIMARY KEY(user_id, friend_id))

CREATE TABLE likes(
film_id INTEGER REFERENCES films(id),
user_id INTEGER REFERENCES users(id),
PRIMARY KEY(film_id, user_id)
)
